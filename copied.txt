import sys
import serial
import time

# Set your VM mapped serial device
SERIAL_PORT = '/dev/ttyS0'  
BAUD_RATE = 9600

try:
    ser = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=1)
    print("Connected to SimulIDE Serial System")
except Exception as e:
    print(f"Error: {e}")
    ser = None

def control_pump(direction, speed_percentage):
    """
    direction: 'F' (Forward/Clockwise) or 'R' (Reverse/CCW)
    speed_percentage: 0 to 100
    """
    if not ser or not ser.is_open:
        return

    # Convert 0-100% speed to 0-255 raw PWM value
    pwm_value = int((speed_percentage / 100.0) * 255)
    
    # Construct command: M,F,255\n
    command = f"M,{direction},{pwm_value}\n"
    
    # Send it over serial to the Arduino
    ser.write(command.encode('utf-8'))
    print(f"Sent command to Pump: {command.strip()} ({speed_percentage}% speed)")

def main():
    # sys.argv[0] is the script name itself, sys.argv[1:] gets all following arguments
    args = sys.argv[1:]
    
    if not args:
        print("Error: No arguments provided.")
        print("Usage example: python pump_control.py 50,3,c 40,2,cc")
        return

    if ser:
        time.sleep(2) # Allow Arduino simulator boot time
        
        try:
            for block in args:
                # Split each block by the comma delimiter
                parts = block.split(',')
                if len(parts) != 3:
                    print(f"Skipping invalid format: '{block}'. Must be speed,duration,rotation")
                    continue
                
                # Parse values
                speed = int(parts[0])
                duration = float(parts[1])
                rotation_raw = parts[2].lower()
                
                # Map 'c' to 'F' and 'cc' to 'R'
                if rotation_raw == 'c':
                    direction = 'F'
                    rot_name = "Clockwise"
                elif rotation_raw == 'cc':
                    direction = 'R'
                    rot_name = "Counter-Clockwise"
                else:
                    print(f"Skipping invalid rotation '{rotation_raw}' in block '{block}'. Use 'c' or 'cc'.")
                    continue
                
                print(f"\n--- Running: Speed={speed}%, Duration={duration}s, Direction={rot_name} ---")
                
                # Execute command
                control_pump(direction, speed)
                
                # Hold for the specified duration
                time.sleep(duration)
                
        except KeyboardInterrupt:
            print("\nSequence interrupted by user.")
        finally:
            # Safety Shutdown: Always stop the pump when the sequence finishes or breaks
            print("\nExecuting safety shutdown...")
            control_pump('F', 0)
            ser.close()
            print("Serial connection closed.")

if __name__ == '__main__':
    main()
